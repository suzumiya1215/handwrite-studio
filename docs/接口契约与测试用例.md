### 0.3 `fake_font.py` — M7 开发的命脉

**为什么必须先写这个**：没有真实手写样本时，`FillService` 无法调试。
用程序化生成的假字形，测试**瞬间完成**且**完全确定**。

```python
"""程序化生成一个 12 字符的假字形库，供 FillService 测试用。"""
from PIL import Image, ImageDraw
from core.models.glyph import FontLibrary, GlyphSet, GlyphSample
from core.base.charset import CHARSET_V1

def _render_glyph(ch: str, variant: int, h: int = 128) -> Image.Image:
    """
    把一个字符画成 RGBA 图（黑字透明底），高度 h。
    variant 控制形状微差（模拟同一字符的不同手写样本）。
    """
    w = h if ch not in ",." else max(12, h // 9)     # 标点窄
    img = Image.new("RGBA", (w, h), (0, 0, 0, 0))
    d = ImageDraw.Draw(img)
    # 用内置字体画（不依赖任何系统字体，CI 可跑）
    d.text((2, 2), ch, fill=(0, 0, 0, 255))
    # variant 让每个样本略微不同
    if variant:
        img = img.rotate(variant * 2.0, resample=Image.BICUBIC, expand=False)
    return img

def make_fake_font(samples_per_char: int = 2) -> FontLibrary:
    """返回一个完整的、可直接喂给 FillService 的 FontLibrary。"""
    chars = {}
    for ch in CHARSET_V1:
        samples = []
        for i in range(samples_per_char):
            g = _render_glyph(ch, variant=i)
            samples.append(GlyphSample(
                file=f"{i+1:03d}.png", width=g.width, height=g.height, _image=g
            ))
        chars[ch] = GlyphSet(char=ch, samples=samples)
    return FontLibrary(
        id="test-font", name="测试字迹", charset=["digits", "punct"],
        glyph_height=128, characters=chars,
        created_at="2026-10-09T00:00:00+08:00",
        updated_at="2026-10-09T00:00:00+08:00",
    )

def make_blind_font(missing: set[str] | None = None) -> FontLibrary:
    """缺字版本：用来测「跳过字符」的计数是否正确。"""
    lib = make_fake_font()
    for ch in (missing or {"A"}):        # 移除几个字符
        lib.characters.pop(ch, None)
    return lib
```

class TestMmPx:
    def test_identity_at_25_4dpi(self):
        """25.4 DPI 下，1 英寸 = 1 mm → 恰好相等。"""
        assert G.mm_to_px(MM_PER_INCH := 25.4, dpi=25) == pytest.approx(25.0)

    def test_a4_at_300dpi(self):
        """A4 宽 210mm @300dpi = 210/25.4*300 ≈ 2480.31"""

import pytest
from core.base import charset as C

class TestCharsetV1:
    def test_exactly_twelve(self):
        """★ v1 字符集恰好 12 个。"""
        assert len(C.CHARSET_V1) == 12

    def test_composition(self):
        assert C.CHARSET_V1 == list("0123456789") + [",", "."]

        """顺序固定：先数字后标点。改了会让字形库目录顺序变化。"""

        assert C.CHARSET_V1[10] == ","

class TestDirnameMapping:
    def test_digits_map_to_themselves(self):
        for d in "0123456789":
            assert C.char_to_dirname(d) == d

    def test_punct_map_to_words(self):
        assert C.char_to_dirname(",") == "comma"
        assert C.char_to_dirname(".") == "dot"

    def test_dirname_to_char_roundtrip(self):
        """★ C11：双向映射必须自洽。"""
        for ch in C.CHARSET_V1:
            assert C.dirname_to_char(C.char_to_dirname(ch)) == ch

    def test_all_dirnames_ascii(self):

            C.char_to_dirname(ch).encode("ascii")   # 不抛即通过

    def test_unknown_char_raises(self):

            C.char_to_dirname("你")

        assert not C.is_supported("A")

## 3. M1 · `imageops` 契约测试

**契约依据**：模块设计说明书 §4.1

```python
# tests/test_imageops.py
import numpy as np
import pytest
from PIL import Image
from core.base import imageops as I

def make_grid_photo(rows=3, cols=5, cell=60, margin=20) -> np.ndarray:
    """
    合成一张「手写数字照片」：白底 + 若干黑方块。
    方块尺寸可控，用于测 find_boxes 的计数与排序。
    """
    H = margin*2 + rows*cell
    W = margin*2 + cols*cell
    img = np.full((H, W, 3), 255, np.uint8)
    for r in range(rows):
        for c in range(cols):
            y0 = margin + r*cell + cell//4
            x0 = margin + c*cell + cell//4
            img[y0:y0+cell//2, x0:x0+cell//2] = 0
    return img

class TestBinarize:
    def test_foreground_is_255(self):
        img = make_grid_photo()
        b = I.binarize(img, tolerance=29)
        assert b.dtype == np.uint8
        assert set(np.unique(b)).issubset({0, 255})
        assert b[30, 30] == 255            # 黑方块 → 前景
        assert b[0, 0] == 0                # 白底 → 背景

    def test_invert(self):
        img = make_grid_photo()
        b = I.binarize(img, tolerance=29, invert=True)
        assert b[0, 0] == 255

    def test_tolerance_monotonic(self):
        """容差越大，保留的前景像素越多（单调不减）。"""
        img = make_grid_photo()
        n_low = int((I.binarize(img, 10) > 0).sum())
        n_high = int((I.binarize(img, 90) > 0).sum())
        assert n_high >= n_low

    def test_shape_preserved(self):
        img = make_grid_photo()
        assert I.binarize(img).shape == img.shape[:2]

class TestFindBoxes:
    def test_counts_all_squares(self):
        img = make_grid_photo(rows=3, cols=5)
        boxes = I.find_boxes(I.binarize(img))
        assert len(boxes) == 15

    def test_sorted_row_major(self):
        """★ 排序必须是「先上后下、同排先左后右」，否则标注全乱。"""
        img = make_grid_photo(rows=2, cols=3)
        boxes = I.find_boxes(I.binarize(img))
        ys = [b[1] for b in boxes]
        # 前 3 个的 y 应明显小于后 3 个
        assert max(ys[:3]) < min(ys[3:])
        # 每排内部 x 递增
        assert boxes[0][0] < boxes[1][0] < boxes[2][0]

    def test_filters_noise(self):
        """撒几个 1~2px 噪点 → 应被面积阈值滤掉。"""
        img = make_grid_photo(rows=1, cols=1)
        img[0:2, 0:2] = 0
        boxes = I.find_boxes(I.binarize(img))
        assert len(boxes) == 1

    def test_filters_table_lines(self):
        """画一条很长的竖线 → 长宽比过滤应剔除。"""
        img = np.full((300, 300, 3), 255, np.uint8)
        img[20:280, 100:104] = 0          # 细长竖线
        boxes = I.find_boxes(I.binarize(img))
        assert len(boxes) == 0

    def test_small_punct_not_killed(self):
        """★ 陷阱用例：小圆点必须被保留（动态面积阈值）。"""
        img = np.full((400, 400, 3), 255, np.uint8)
        # 一个正常数字
        img[50:150, 50:130] = 0
        # 一个小句点（约字高的 12%）
        img[250:262, 55:67] = 0
        boxes = I.find_boxes(I.binarize(img))
        assert len(boxes) == 2

class TestMergeNearby:
    def test_merges_two_dots_of_colon(self):
        """两个相距很近的点 → 合并成一个框。"""
        boxes = [(10, 10, 8, 8), (10, 20, 8, 8)]
        m = I.merge_nearby_boxes(boxes, gap_px=4)
        assert len(m) == 1

    def test_keeps_far_apart(self):
        boxes = [(10, 10, 8, 8), (10, 90, 8, 8)]
        assert len(I.merge_nearby_boxes(boxes, gap_px=4)) == 2

class TestNormalizeGlyph:
    def test_output_height(self):
        img = make_grid_photo()
        b = I.binarize(img)
        box = I.find_boxes(b)[0]
        g, (w0, h0) = I.normalize_glyph(img, box, target_height=128)
        assert g.height == 128
        assert g.mode == "RGBA"

    def test_foreground_black_bg_transparent(self):
        """★ C12：前景纯黑、背景全透明。"""
        img = make_grid_photo()
        b = I.binarize(img)
        box = I.find_boxes(b)[0]
        g, _ = I.normalize_glyph(img, box)
        a = np.array(g)
        assert a[..., 3].max() == 255        # 有完全不透明处
        assert a[..., 3].min() == 0          # 有全透明处
        opaque = a[a[..., 3] > 200]
        assert (opaque[:, :3] < 30).all()    # 不透明处接近纯黑

    def test_aspect_ratio_preserved(self):
        """宽高比变化不超过 5%。"""
        img = make_grid_photo()
        b = I.binarize(img)
        box = I.find_boxes(b)[0]
        g, (w0, h0) = I.normalize_glyph(img, box)
        assert abs(g.width / g.height - w0 / h0) < 0.05

    def test_returns_original_size(self):
        """第二个返回值是原始尺寸，供 sample_sizes 用。"""
        img = make_grid_photo()
        b = I.binarize(img)
        box = I.find_boxes(b)[0]
        _, (w0, h0) = I.normalize_glyph(img, box)
        assert (w0, h0) == (box[2], box[3])

    def test_has_transparent_padding(self):
        """四边留白 ≥2px（防旋转裁笔画）。"""
        img = make_grid_photo()
        b = I.binarize(img)
        box = I.find_boxes(b)[0]
        g, _ = I.normalize_glyph(img, box, pad_px=2)
        a = np.array(g)[..., 3]
        assert a[0].max() == 0 and a[-1].max() == 0
        assert a[:, 0].max() == 0 and a[:, -1].max() == 0

class TestChinesePath:
    def test_load_image_any_with_chinese(self, tmp_path):
        """★ cv2.imread 对中文路径返回 None，必须用 fromfile+imdecode。"""
        p = tmp_path / "我的字迹照片.png"
        Image.new("RGB", (100, 100), (255, 255, 255)).save(p)
        arr = I.load_image_any(str(p))
        assert arr is not None and arr.shape == (100, 100, 3)

class TestShape:
    def test_scale(self):
        g = Image.new("RGBA", (100, 50), (0, 0, 0, 255))
        out = I.scale(g, 2.0, 2.0)
        assert out.size == (200, 100)

    def test_rotate_expands(self):
        g = Image.new("RGBA", (100, 100), (0, 0, 0, 255))
        out = I.rotate(g, 45)
        assert out.width > 100

    def test_slant_is_not_rotation(self):
        """★ 倾斜用错切：高不变、宽增加；旋转会两者都变。"""
        g = Image.new("RGBA", (100, 100), (0, 0, 0, 255))
        out = I.slant(g, 10)
        assert out.height == 100
        assert out.width > 100

    def test_dilate_only_alpha(self):
        """加粗不应改变 RGB（保持纯黑），只扩张 alpha。"""
        g = Image.new("RGBA", (50, 50), (0, 0, 0, 0))
        g.putpixel((25, 25), (0, 0, 0, 255))
        out = I.dilate_alpha(g, 2.0)
        assert np.array(out)[..., 3].sum() > np.array(g)[..., 3].sum()
        assert (np.array(out)[..., 3] > 0)[..., None].any()
```

# tests/test_layering.py
"""架构分层规则的自动校验。这些测试不需要导入被测模块，只做静态扫描。"""
import re
from pathlib import Path
import pytest

ROOT = Path(__file__).resolve().parents[1]
CORE = ROOT / "core"

def py_files(sub: str) -> list[Path]:
    d = CORE / sub
    return list(d.rglob("*.py")) if d.exists() else []

class TestNoQtInCore:
    def test_no_pyside_import(self):
        """★ N1：core/ 下不得 import PySide6。"""
        bad = []
        for f in py_files(""):
            t = f.read_text(encoding="utf-8")
            if re.search(r"^\s*(import|from)\s+PySide6", t, re.M):
                bad.append(f.relative_to(ROOT))
        assert not bad, f"core/ 下出现 Qt 依赖: {bad}"

    def test_no_qmessagebox(self):
        """★ N2：core/ 不得弹窗。"""
        bad = [f for f in py_files("") if "QMessageBox" in f.read_text(encoding="utf-8")]
        assert not bad, f"core/ 下出现弹窗: {bad}"

class TestNoHardcoding:
    def test_no_raw_25_4(self):
        """★ N3：25.4 只允许出现在 geometry.py。"""
        bad = []
        for f in py_files(""):
            if f.name == "geometry.py":
                continue
            if "25.4" in f.read_text(encoding="utf-8"):
                bad.append(f.relative_to(ROOT))
        assert not bad, f"硬编码 25.4: {bad}"

    def test_no_raw_dirnames(self):
        """★ N4：comma/dot 目录名只能由 charset 生成。"""
        bad = []
        for f in py_files(""):
            if f.name == "charset.py":
                continue
            t = f.read_text(encoding="utf-8")
            if re.search(r'["\'](comma|dot)["\']', t):
                bad.append(f.relative_to(ROOT))
        assert not bad, f"硬编码字符目录名: {bad}"

    def test_no_stdlib_random(self):
        """★ N5：统一用 np.random.default_rng。"""
        bad = []
        for f in py_files(""):
            t = f.read_text(encoding="utf-8")
            if re.search(r"^\s*import\s+random\b", t, re.M):
                bad.append(f.relative_to(ROOT))
        assert not bad, f"出现 import random: {bad}"

    def test_no_global_np_seed(self):
        """★ N6：np.random.seed 会破坏可复现性。"""
        bad = []
        for f in py_files(""):
            if "np.random.seed" in f.read_text(encoding="utf-8"):
                bad.append(f.relative_to(ROOT))
        assert not bad, f"出现全局播种: {bad}"
